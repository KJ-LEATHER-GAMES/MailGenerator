Option Explicit

' ============================================================
' Technical Spike A v0.2
'
' Route H : HTMLBody
' Route W : WordEditor
'
' 対象:
'   TSA-001 ～ TSA-010
'
' 注意:
'   - 本番実装ではなくTechnical Spike用
'   - Outlook / WordともLate Binding
'   - 参照設定不要
'   - Route H / Wは別MailItemで検証
' ============================================================


' ------------------------------------------------------------
' ★ 実環境に合わせて変更
' ------------------------------------------------------------
Private Const OFT_PATH As String = _
    "C:\Temp\TechnicalSpikeA.oft"


' ------------------------------------------------------------
' Outlook constants
' ------------------------------------------------------------
Private Const OL_DISCARD As Long = 1
Private Const OL_FOLDER_DRAFTS As Long = 16


' ------------------------------------------------------------
' Word constants
' Late Binding用
' ------------------------------------------------------------
Private Const WD_FIND_STOP As Long = 0


' ------------------------------------------------------------
' Result counters
' ------------------------------------------------------------
Private mPassH As Long
Private mFailH As Long
Private mManualH As Long

Private mPassW As Long
Private mFailW As Long
Private mManualW As Long


' ============================================================
' Entry Point
' ============================================================
Public Sub RunTechnicalSpikeA_v02()

    Dim olApp As Object

    ' ----- Scan用 -----
    Dim scanItemH As Object
    Dim scanItemW As Object

    Dim scanInspectorW As Object
    Dim scanWordDocW As Object

    Dim subjectH As String
    Dim htmlH As String

    Dim subjectW As String
    Dim wordTextW As String

    Dim varsH As Object
    Dim locationsH As Object
    Dim invalidH As Collection

    Dim varsW As Object
    Dim locationsW As Object
    Dim invalidW As Collection

    Dim wordScanAvailable As Boolean

    ' ----- Lifecycle -----
    Dim draftsBefore As Long
    Dim draftsAfter As Long

    Dim inspectorsBefore As Long
    Dim inspectorsAfter As Long

    Dim closeErrH As Long
    Dim closeErrW As Long

    Dim sideEffectAnswer As VbMsgBoxResult
    Dim lifecycleVisualOK As Boolean


    ' ----- Render用 -----
    Dim renderItemH As Object
    Dim renderItemW As Object

    Dim renderInspectorW As Object
    Dim renderWordDocW As Object

    Dim originalSubjectH As String
    Dim originalHtmlH As String

    Dim originalSubjectW As String
    Dim originalWordTextW As String

    Dim renderedSubjectH As String
    Dim renderedHtmlH As String

    Dim renderedSubjectW As String

    Dim values As Object

    Dim wordRenderAvailable As Boolean

    Dim replaceCountH As Long
    Dim replaceCountW As Long


    ' ----- Unresolved -----
    Dim unresolvedFullH As Long
    Dim unresolvedFullW As Long

    Dim unresolvedNegativeH As Long
    Dim unresolvedNegativeW As Long


    ' ----- Negative test用 -----
    Dim negativeItemH As Object
    Dim negativeItemW As Object

    Dim negativeInspectorW As Object
    Dim negativeWordDocW As Object

    Dim negativeSubjectH As String
    Dim negativeHtmlH As String

    Dim negativeSubjectW As String

    Dim negativeWordAvailable As Boolean


    ' ----- Display / Manual -----
    Dim displayErrH As Long
    Dim displayErrW As Long

    Dim visualH As VbMsgBoxResult
    Dim visualW As VbMsgBoxResult


    On Error GoTo FatalError

    ResetCounters

    PrintHeader


    ' ========================================================
    ' Precondition
    ' ========================================================
    If Len(Dir$(OFT_PATH)) = 0 Then

        Debug.Print "[FATAL] OFT not found:"
        Debug.Print OFT_PATH

        Exit Sub

    End If


    Set olApp = GetOutlookApplication()

    If olApp Is Nothing Then

        Debug.Print _
            "[FATAL] Outlook.Application could not be created."

        Exit Sub

    End If


    draftsBefore = GetDraftCount(olApp)
    inspectorsBefore = GetInspectorCount(olApp)

    Debug.Print
    Debug.Print "Drafts before     : " & draftsBefore
    Debug.Print "Inspectors before : " & inspectorsBefore


    ' ========================================================
    ' TSA-001
    ' OFT読込
    ' ========================================================

    Set scanItemH = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)

    Set scanItemW = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)


    LogRouteResult _
        "TSA-001", _
        "H", _
        Not scanItemH Is Nothing, _
        "Route H用MailItemを生成"


    LogRouteResult _
        "TSA-001", _
        "W", _
        Not scanItemW Is Nothing, _
        "Route W用MailItemを生成"


    ' ========================================================
    ' TSA-002
    ' Body Access
    ' ========================================================

    ' --------------------------------------------------------
    ' Route H
    ' --------------------------------------------------------
    If Not scanItemH Is Nothing Then

        subjectH = SafeString(scanItemH.Subject)
        htmlH = SafeString(scanItemH.HTMLBody)

        LogRouteResult _
            "TSA-002", _
            "H", _
            (Len(subjectH) > 0 And Len(htmlH) > 0), _
            "Subject / HTMLBody取得"

        Debug.Print
        Debug.Print "----- Route H -----"
        Debug.Print "Subject:"
        Debug.Print subjectH
        Debug.Print "HTMLBody length: " & Len(htmlH)

    Else

        LogRouteResult _
            "TSA-002", _
            "H", _
            False, _
            "TSA-001失敗のためBody Access不可"

    End If


    ' --------------------------------------------------------
    ' Route W
    ' --------------------------------------------------------
    If Not scanItemW Is Nothing Then

        subjectW = SafeString(scanItemW.Subject)

        wordScanAvailable = _
            TryGetWordEditor( _
                scanItemW, _
                scanInspectorW, _
                scanWordDocW)

        If wordScanAvailable Then

            wordTextW = _
                SafeString(scanWordDocW.Content.Text)

        End If

        LogRouteResult _
            "TSA-002", _
            "W", _
            (wordScanAvailable And _
             Len(subjectW) > 0 And _
             Len(wordTextW) > 0), _
            "Subject / Inspector / WordEditor / Word Document取得"

        Debug.Print
        Debug.Print "----- Route W -----"
        Debug.Print "Subject:"
        Debug.Print subjectW

        If wordScanAvailable Then
            Debug.Print _
                "Word Content length: " & Len(wordTextW)
        Else
            Debug.Print _
                "WordEditor unavailable before Display."
        End If

    Else

        LogRouteResult _
            "TSA-002", _
            "W", _
            False, _
            "TSA-001失敗のためWordEditor Access不可"

    End If


    ' ========================================================
    ' TSA-003
    ' Variable検出・表現
    ' ========================================================

    If Not scanItemH Is Nothing Then

        LogRouteResult _
            "TSA-003", _
            "H", _
            CheckExpectedBodyTokens( _
                htmlH, _
                "Route H / HTMLBody"), _
            "HTMLBody上でVariableが連続文字列として存在"

    Else

        LogRouteResult _
            "TSA-003", _
            "H", _
            False, _
            "Route H MailItemなし"

    End If


    If wordScanAvailable Then

        LogRouteResult _
            "TSA-003", _
            "W", _
            CheckExpectedBodyTokens( _
                wordTextW, _
                "Route W / WordEditor"), _
            "Word Document上でVariableを検索可能"

    Else

        LogRouteResult _
            "TSA-003", _
            "W", _
            False, _
            "WordEditor取得不可"

    End If


    ' ========================================================
    ' TSA-004
    ' Scanner / Case Insensitive
    ' ========================================================

    ' --------------------------------------------------------
    ' Route H
    ' --------------------------------------------------------
    If Len(subjectH) > 0 And Len(htmlH) > 0 Then

        ScanTemplateText _
            subjectH, _
            htmlH, _
            varsH, _
            locationsH, _
            invalidH

        PrintScanResult _
            "Route H", _
            varsH, _
            locationsH, _
            invalidH

        LogRouteResult _
            "TSA-004", _
            "H", _
            VerifyExpectedScan( _
                varsH, _
                locationsH, _
                invalidH), _
            "HTMLBody Scanner / Case Insensitive / Location"

    Else

        LogRouteResult _
            "TSA-004", _
            "H", _
            False, _
            "Scanner入力取得不可"

    End If


    ' --------------------------------------------------------
    ' Route W
    ' --------------------------------------------------------
    If wordScanAvailable Then

        ScanTemplateText _
            subjectW, _
            wordTextW, _
            varsW, _
            locationsW, _
            invalidW

        PrintScanResult _
            "Route W", _
            varsW, _
            locationsW, _
            invalidW

        LogRouteResult _
            "TSA-004", _
            "W", _
            VerifyExpectedScan( _
                varsW, _
                locationsW, _
                invalidW), _
            "WordEditor Scanner / Case Insensitive / Location"

    Else

        LogRouteResult _
            "TSA-004", _
            "W", _
            False, _
            "WordEditor Scannerを実施できず"

    End If


    ' ========================================================
    ' TSA-010
    ' Scan用MailItem Lifecycle
    '
    ' Render前にScan MailItemを破棄する。
    ' ========================================================

    On Error Resume Next


    If Not scanItemH Is Nothing Then

        Err.Clear

        scanItemH.Close OL_DISCARD

        closeErrH = Err.Number

    Else

        closeErrH = -1

    End If


    If Not scanItemW Is Nothing Then

        Err.Clear

        scanItemW.Close OL_DISCARD

        closeErrW = Err.Number

    Else

        closeErrW = -1

    End If


    Set scanWordDocW = Nothing
    Set scanInspectorW = Nothing

    Set scanItemH = Nothing
    Set scanItemW = Nothing


    Err.Clear

    On Error GoTo FatalError


    DoEvents


    draftsAfter = GetDraftCount(olApp)
    inspectorsAfter = GetInspectorCount(olApp)

    Debug.Print
    Debug.Print "Drafts after scan     : " & draftsAfter
    Debug.Print "Inspectors after scan : " & inspectorsAfter


    ' --------------------------------------------------------
    ' VBAから完全判定できない
    ' 「画面が勝手に開かなかったか」を確認
    ' --------------------------------------------------------
    sideEffectAnswer = MsgBox( _
        "ここまでがScan処理です。" & vbCrLf & vbCrLf & _
        "この時点ではRender結果のメールを" & _
        "意図的にはDisplayしていません。" & vbCrLf & vbCrLf & _
        "Scan中にOutlookのメール編集画面や" & _
        "保存確認ダイアログが勝手に表示されましたか？" & _
        vbCrLf & vbCrLf & _
        "Yes = 表示された" & vbCrLf & _
        "No  = 表示されていない", _
        vbYesNo + vbQuestion, _
        "TSA-010 Lifecycle確認")

    lifecycleVisualOK = _
        (sideEffectAnswer = vbNo)


    LogRouteResult _
        "TSA-010", _
        "H", _
        (closeErrH = 0 And _
         draftsAfter = draftsBefore And _
         lifecycleVisualOK), _
        "Scan用MailItemをDisplayせず破棄"


    LogRouteResult _
        "TSA-010", _
        "W", _
        (closeErrW = 0 And _
         draftsAfter = draftsBefore And _
         inspectorsAfter = inspectorsBefore And _
         lifecycleVisualOK), _
        "WordEditor取得後もMailItemを安全に破棄"


    ' ========================================================
    ' Render用MailItem生成
    ' ========================================================

    Set renderItemH = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)

    Set renderItemW = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)


    If renderItemH Is Nothing Then

        Debug.Print _
            "[Route H] Render MailItem creation failed."

    End If


    If renderItemW Is Nothing Then

        Debug.Print _
            "[Route W] Render MailItem creation failed."

    End If


    Set values = CreateTestValues()


    ' ========================================================
    ' Route H Render preparation
    ' ========================================================

    If Not renderItemH Is Nothing Then

        originalSubjectH = _
            SafeString(renderItemH.Subject)

        originalHtmlH = _
            SafeString(renderItemH.HTMLBody)

        renderedSubjectH = originalSubjectH
        renderedHtmlH = originalHtmlH

    End If


    ' ========================================================
    ' Route W Render preparation
    ' ========================================================

    If Not renderItemW Is Nothing Then

        originalSubjectW = _
            SafeString(renderItemW.Subject)

        wordRenderAvailable = _
            TryGetWordEditor( _
                renderItemW, _
                renderInspectorW, _
                renderWordDocW)

        If wordRenderAvailable Then

            originalWordTextW = _
                SafeString(renderWordDocW.Content.Text)

        End If

        renderedSubjectW = originalSubjectW

    End If


    ' ========================================================
    ' TSA-005
    ' Body置換比較
    ' ========================================================

    ' --------------------------------------------------------
    ' Route H
    ' --------------------------------------------------------
    If Not renderItemH Is Nothing Then

        RenderSubject _
            renderedSubjectH, _
            values, _
            vbNullString

        replaceCountH = _
            RenderHtmlBody( _
                renderedHtmlH, _
                values, _
                vbNullString)

        renderItemH.Subject = renderedSubjectH
        renderItemH.HTMLBody = renderedHtmlH

        LogRouteResult _
            "TSA-005", _
            "H", _
            VerifyTokenReplacement( _
                originalSubjectH, _
                originalHtmlH, _
                renderedSubjectH, _
                renderedHtmlH, _
                values), _
            "HTMLBody局所置換 / replacements=" & _
            replaceCountH

    Else

        LogRouteResult _
            "TSA-005", _
            "H", _
            False, _
            "Render MailItemなし"

    End If


    ' --------------------------------------------------------
    ' Route W
    ' --------------------------------------------------------
    If wordRenderAvailable Then

        RenderSubject _
            renderedSubjectW, _
            values, _
            vbNullString

        renderItemW.Subject = renderedSubjectW

        replaceCountW = _
            RenderWordBody( _
                renderWordDocW, _
                values, _
                vbNullString)

        LogRouteResult _
            "TSA-005", _
            "W", _
            VerifyTokenReplacement( _
                originalSubjectW, _
                originalWordTextW, _
                renderedSubjectW, _
                SafeString(renderWordDocW.Content.Text), _
                values), _
            "Word Range置換 / replacements=" & _
            replaceCountW

    Else

        LogRouteResult _
            "TSA-005", _
            "W", _
            False, _
            "WordEditor Render不可"

    End If


    ' ========================================================
    ' TSA-006
    ' 特殊文字安全性
    '
    ' ModelName = A&B <TEST>
    ' ========================================================

    ' --------------------------------------------------------
    ' Route H
    ' HTML Encodeされた文字列がHTML上にあるか
    ' --------------------------------------------------------
    If Not renderItemH Is Nothing Then

        LogRouteResult _
            "TSA-006", _
            "H", _
            VerifyHtmlEncodedValue( _
                SafeString(renderItemH.HTMLBody), _
                "A&B <TEST>"), _
            "HTML Encode後にHTMLBodyへ設定"

    Else

        LogRouteResult _
            "TSA-006", _
            "H", _
            False, _
            "Route H Render不可"

    End If


    ' --------------------------------------------------------
    ' Route W
    ' Raw文字列としてWord文書上に存在するか
    ' --------------------------------------------------------
    If wordRenderAvailable Then

        LogRouteResult _
            "TSA-006", _
            "W", _
            (InStr( _
                1, _
                SafeString(renderWordDocW.Content.Text), _
                "A&B <TEST>", _
                vbBinaryCompare) > 0), _
            "HTML EncodeなしでWord Rangeへ文字列設定"

    Else

        LogRouteResult _
            "TSA-006", _
            "W", _
            False, _
            "WordEditor Render不可"

    End If


    ' ========================================================
    ' TSA-008
    ' Unresolved Check
    ' ========================================================

    ' --------------------------------------------------------
    ' Full Render
    ' --------------------------------------------------------
    If Not renderItemH Is Nothing Then

        unresolvedFullH = _
            CountValidUnresolved( _
                SafeString(renderItemH.Subject)) + _
            CountValidUnresolved( _
                SafeString(renderItemH.HTMLBody))

    Else

        unresolvedFullH = -1

    End If


    If wordRenderAvailable Then

        unresolvedFullW = _
            CountValidUnresolved( _
                SafeString(renderItemW.Subject)) + _
            CountValidUnresolved( _
                SafeString(renderWordDocW.Content.Text))

    Else

        unresolvedFullW = -1

    End If


    ' --------------------------------------------------------
    ' Negative Route H
    ' GateDateのみ意図的に未置換
    ' --------------------------------------------------------
    Set negativeItemH = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)

    If Not negativeItemH Is Nothing Then

        negativeSubjectH = _
            SafeString(negativeItemH.Subject)

        negativeHtmlH = _
            SafeString(negativeItemH.HTMLBody)

        RenderSubject _
            negativeSubjectH, _
            values, _
            "GateDate"

        Call RenderHtmlBody( _
            negativeHtmlH, _
            values, _
            "GateDate")

        negativeItemH.Subject = negativeSubjectH
        negativeItemH.HTMLBody = negativeHtmlH

        unresolvedNegativeH = _
            CountValidUnresolved(negativeSubjectH) + _
            CountValidUnresolved(negativeHtmlH)

    Else

        unresolvedNegativeH = -1

    End If


    ' --------------------------------------------------------
    ' Negative Route W
    ' --------------------------------------------------------
    Set negativeItemW = _
        TryCreateItemFromTemplate(olApp, OFT_PATH)

    If Not negativeItemW Is Nothing Then

        negativeWordAvailable = _
            TryGetWordEditor( _
                negativeItemW, _
                negativeInspectorW, _
                negativeWordDocW)

        If negativeWordAvailable Then

            negativeSubjectW = _
                SafeString(negativeItemW.Subject)

            RenderSubject _
                negativeSubjectW, _
                values, _
                "GateDate"

            negativeItemW.Subject = negativeSubjectW

            Call RenderWordBody( _
                negativeWordDocW, _
                values, _
                "GateDate")

            unresolvedNegativeW = _
                CountValidUnresolved(negativeSubjectW) + _
                CountValidUnresolved( _
                    SafeString(negativeWordDocW.Content.Text))

        Else

            unresolvedNegativeW = -1

        End If

    Else

        unresolvedNegativeW = -1

    End If


    Debug.Print
    Debug.Print "----- Unresolved Check -----"
    Debug.Print _
        "Route H full     : " & unresolvedFullH
    Debug.Print _
        "Route H negative : " & unresolvedNegativeH
    Debug.Print _
        "Route W full     : " & unresolvedFullW
    Debug.Print _
        "Route W negative : " & unresolvedNegativeW


    LogRouteResult _
        "TSA-008", _
        "H", _
        (unresolvedFullH = 0 And _
         unresolvedNegativeH > 0), _
        "Full=0 / GateDate未置換=>0"


    LogRouteResult _
        "TSA-008", _
        "W", _
        (unresolvedFullW = 0 And _
         unresolvedNegativeW > 0), _
        "Full=0 / GateDate未置換=>0"


    ' Negative MailItemを破棄
    On Error Resume Next

    If Not negativeItemH Is Nothing Then
        negativeItemH.Close OL_DISCARD
    End If

    If Not negativeItemW Is Nothing Then
        negativeItemW.Close OL_DISCARD
    End If

    Set negativeWordDocW = Nothing
    Set negativeInspectorW = Nothing

    Set negativeItemH = Nothing
    Set negativeItemW = Nothing

    On Error GoTo FatalError


    ' ========================================================
    ' TSA-009 + TSA-007
    '
    ' Route Hを先に表示・確認
    ' Route Wを次に表示・確認
    ' ========================================================

    ' --------------------------------------------------------
    ' Route H Display
    ' --------------------------------------------------------
    If Not renderItemH Is Nothing Then

        On Error Resume Next

        Err.Clear

        renderItemH.Display

        displayErrH = Err.Number

        On Error GoTo FatalError


        LogRouteResult _
            "TSA-009", _
            "H", _
            (displayErrH = 0), _
            "Route H MailItem.Display"

    Else

        displayErrH = -1

        LogRouteResult _
            "TSA-009", _
            "H", _
            False, _
            "Route H Render MailItemなし"

    End If


    If displayErrH = 0 Then

        DoEvents

        visualH = AskFormattingResult( _
            "Route H：HTMLBody")

        Select Case visualH

            Case vbYes

                LogRouteResult _
                    "TSA-007", _
                    "H", _
                    True, _
                    "基本書式維持を目視確認"

            Case vbNo

                LogRouteResult _
                    "TSA-007", _
                    "H", _
                    False, _
                    "基本書式維持に問題あり"

            Case Else

                LogRouteManual _
                    "TSA-007", _
                    "H", _
                    "目視確認未完了"

        End Select

    Else

        LogRouteResult _
            "TSA-007", _
            "H", _
            False, _
            "Display不可のため目視確認不能"

    End If


    ' --------------------------------------------------------
    ' Route W Display
    ' --------------------------------------------------------
    If Not renderItemW Is Nothing Then

        On Error Resume Next

        Err.Clear

        renderItemW.Display

        displayErrW = Err.Number

        On Error GoTo FatalError


        LogRouteResult _
            "TSA-009", _
            "W", _
            (displayErrW = 0), _
            "Route W MailItem.Display"

    Else

        displayErrW = -1

        LogRouteResult _
            "TSA-009", _
            "W", _
            False, _
            "Route W Render MailItemなし"

    End If


    If displayErrW = 0 Then

        DoEvents

        visualW = AskFormattingResult( _
            "Route W：WordEditor")

        Select Case visualW

            Case vbYes

                LogRouteResult _
                    "TSA-007", _
                    "W", _
                    True, _
                    "基本書式維持を目視確認"

            Case vbNo

                LogRouteResult _
                    "TSA-007", _
                    "W", _
                    False, _
                    "基本書式維持に問題あり"

            Case Else

                LogRouteManual _
                    "TSA-007", _
                    "W", _
                    "目視確認未完了"

        End Select

    Else

        LogRouteResult _
            "TSA-007", _
            "W", _
            False, _
            "Display不可のため目視確認不能"

    End If


    ' ========================================================
    ' Final result
    ' ========================================================

    PrintFinalResult

    Debug.Print
    Debug.Print _
        "NOTE:"
    Debug.Print _
        "PASS件数だけでRouteを選定しないこと。"
    Debug.Print _
        "v0.2仕様書 §13 の判断軸に従って比較すること。"


Cleanup:

    ' Displayした2通はEvidence確認のため、
    ' ここでは自動Closeしない。
    '
    ' 検証後、送信せず手動で閉じる。

    Set values = Nothing

    Set varsH = Nothing
    Set locationsH = Nothing
    Set invalidH = Nothing

    Set varsW = Nothing
    Set locationsW = Nothing
    Set invalidW = Nothing

    Set renderWordDocW = Nothing
    Set renderInspectorW = Nothing

    Set renderItemH = Nothing
    Set renderItemW = Nothing

    Set olApp = Nothing

    Exit Sub


FatalError:

    Debug.Print
    Debug.Print "========================================"
    Debug.Print "[FATAL ERROR]"
    Debug.Print "Number      : " & Err.Number
    Debug.Print "Description : " & Err.Description
    Debug.Print "========================================"

    On Error Resume Next

    If Not scanItemH Is Nothing Then
        scanItemH.Close OL_DISCARD
    End If

    If Not scanItemW Is Nothing Then
        scanItemW.Close OL_DISCARD
    End If

    If Not negativeItemH Is Nothing Then
        negativeItemH.Close OL_DISCARD
    End If

    If Not negativeItemW Is Nothing Then
        negativeItemW.Close OL_DISCARD
    End If

    Resume Cleanup

End Sub


' ============================================================
' Outlook
' ============================================================

Private Function GetOutlookApplication() As Object

    Dim app As Object

    On Error Resume Next

    Set app = _
        GetObject(, "Outlook.Application")

    If app Is Nothing Then

        Set app = _
            CreateObject("Outlook.Application")

    End If

    On Error GoTo 0

    Set GetOutlookApplication = app

End Function


Private Function TryCreateItemFromTemplate( _
    ByVal olApp As Object, _
    ByVal oftPath As String) As Object

    Dim item As Object

    On Error Resume Next

    Set item = _
        olApp.CreateItemFromTemplate(oftPath)

    On Error GoTo 0

    Set TryCreateItemFromTemplate = item

End Function


Private Function TryGetWordEditor( _
    ByVal mailItem As Object, _
    ByRef inspector As Object, _
    ByRef wordDocument As Object) As Boolean

    On Error GoTo Failed

    Set inspector = mailItem.GetInspector

    If inspector Is Nothing Then
        GoTo Failed
    End If

    ' Outlook InspectorがWordMailか確認
    If Not inspector.IsWordMail Then
        GoTo Failed
    End If

    Set wordDocument = inspector.WordEditor

    If wordDocument Is Nothing Then
        GoTo Failed
    End If

    TryGetWordEditor = True

    Exit Function


Failed:

    Set wordDocument = Nothing
    Set inspector = Nothing

    TryGetWordEditor = False

End Function


Private Function GetDraftCount( _
    ByVal olApp As Object) As Long

    Dim folder As Object

    On Error GoTo Failed

    Set folder = _
        olApp.Session.GetDefaultFolder(OL_FOLDER_DRAFTS)

    GetDraftCount = folder.Items.Count

    Set folder = Nothing

    Exit Function


Failed:

    GetDraftCount = -1

End Function


Private Function GetInspectorCount( _
    ByVal olApp As Object) As Long

    On Error GoTo Failed

    GetInspectorCount = _
        olApp.Inspectors.Count

    Exit Function


Failed:

    GetInspectorCount = -1

End Function


' ============================================================
' TSA-003
' Variable continuity / visibility
' ============================================================

Private Function CheckExpectedBodyTokens( _
    ByVal bodyText As String, _
    ByVal routeLabel As String) As Boolean

    Dim tokens As Variant
    Dim token As Variant

    Dim found As Boolean
    Dim allFound As Boolean

    tokens = Array( _
        "{{ModelName}}", _
        "{{MODELNAME}}", _
        "{{GateDate}}", _
        "{{Department}}", _
        "{{department}}" _
    )

    allFound = True

    Debug.Print
    Debug.Print _
        "----- TSA-003 " & routeLabel & " -----"

    For Each token In tokens

        found = _
            (InStr( _
                1, _
                bodyText, _
                CStr(token), _
                vbBinaryCompare) > 0)

        Debug.Print _
            CStr(token) & _
            " : " & _
            IIf(found, "FOUND", "NOT FOUND")

        If Not found Then
            allFound = False
        End If

    Next token

    CheckExpectedBodyTokens = allFound

End Function


' ============================================================
' TSA-004
' Common Scanner
'
' Route H:
'   Subject + HTMLBody
'
' Route W:
'   Subject + Word Document.Content.Text
'
' Candidate/Syntaxルールは同じにして比較する。
' ============================================================

Private Sub ScanTemplateText( _
    ByVal subjectText As String, _
    ByVal bodyText As String, _
    ByRef variables As Object, _
    ByRef locations As Object, _
    ByRef invalidCandidates As Collection)

    Set variables = _
        CreateObject("Scripting.Dictionary")

    Set locations = _
        CreateObject("Scripting.Dictionary")

    Set invalidCandidates = _
        New Collection


    variables.CompareMode = vbTextCompare
    locations.CompareMode = vbTextCompare


    ScanOneText _
        subjectText, _
        "Subject", _
        variables, _
        locations, _
        invalidCandidates


    ScanOneText _
        bodyText, _
        "Body", _
        variables, _
        locations, _
        invalidCandidates

End Sub


Private Sub ScanOneText( _
    ByVal sourceText As String, _
    ByVal locationName As String, _
    ByRef variables As Object, _
    ByRef locations As Object, _
    ByRef invalidCandidates As Collection)

    Dim candidateRegex As Object
    Dim syntaxRegex As Object

    Dim matches As Object
    Dim match As Object

    Dim variableName As String
    Dim currentLocation As String


    Set candidateRegex = _
        CreateObject("VBScript.RegExp")

    With candidateRegex

        .Pattern = "\{\{([^{}]*)\}\}"
        .Global = True
        .IgnoreCase = True

    End With


    Set syntaxRegex = _
        CreateObject("VBScript.RegExp")

    With syntaxRegex

        .Pattern = "^[A-Za-z][A-Za-z0-9_]*$"
        .Global = False
        .IgnoreCase = False

    End With


    Set matches = _
        candidateRegex.Execute(sourceText)


    For Each match In matches

        variableName = _
            CStr(match.SubMatches(0))


        If syntaxRegex.Test(variableName) Then

            If Not variables.Exists(variableName) Then

                variables.Add _
                    variableName, _
                    variableName

                locations.Add _
                    variableName, _
                    locationName

            Else

                currentLocation = _
                    CStr(locations(variableName))

                If StrComp( _
                    currentLocation, _
                    locationName, _
                    vbTextCompare) <> 0 Then

                    locations(variableName) = _
                        "Both"

                End If

            End If

        Else

            invalidCandidates.Add _
                locationName & _
                ": {{" & _
                variableName & _
                "}}"

        End If

    Next match

End Sub

    For i = 1 To invalidCandidates.Count

            "  " & _

    Next i

End Sub

' ============================================================
' Subject Renderer
' 共通処理
' ============================================================

Private Sub RenderSubject( _
    ByRef subjectText As String, _
    ByVal values As Object, _
    ByVal skipVariable As String)

    Dim key As Variant
    Dim token As String

    For Each key In values.Keys

        If Len(skipVariable) = 0 Or _
           StrComp( _
               CStr(key), _
               skipVariable, _
               vbTextCompare) <> 0 Then

            token = _
                "{{" & _
                CStr(key) & _
                "}}"

            subjectText = _
                ReplaceAllCI( _
                    subjectText, _
                    token, _
                    CStr(values(key)))

        End If

    Next key

End Sub


' ============================================================
' Route H
' HTMLBody Renderer
' ============================================================

Private Function RenderHtmlBody( _
    ByRef htmlBody As String, _
    ByVal values As Object, _
    ByVal skipVariable As String) As Long

    Dim key As Variant
    Dim token As String

    Dim encodedValue As String

    Dim beforeCount As Long
    Dim totalCount As Long


    For Each key In values.Keys

        If Len(skipVariable) = 0 Or _
           StrComp( _
               CStr(key), _
               skipVariable, _
               vbTextCompare) <> 0 Then


            token = _
                "{{" & _
                CStr(key) & _
                "}}"


            beforeCount = _
                CountOccurrencesCI( _
                    htmlBody, _
                    token)


            encodedValue = _
                HtmlEncodeSpike( _
                    CStr(values(key)))


            htmlBody = _
                ReplaceAllCI( _
                    htmlBody, _
                    token, _
                    encodedValue)


            totalCount = _
                totalCount + _
                beforeCount

        End If

    Next key


    RenderHtmlBody = totalCount

End Function


' ============================================================
' Route W
' WordEditor Renderer
'
' Range.Findで対象Rangeを特定し、
' Range.Textで置換する。
'
' Word Find.ReplaceAllは使用しない。
' ============================================================

Private Function RenderWordBody( _
    ByVal wordDocument As Object, _
    ByVal values As Object, _
    ByVal skipVariable As String) As Long

    Dim key As Variant
    Dim token As String

    Dim count As Long
    Dim totalCount As Long


    For Each key In values.Keys

        If Len(skipVariable) = 0 Or _
           StrComp( _
               CStr(key), _
               skipVariable, _
               vbTextCompare) <> 0 Then


            token = _
                "{{" & _
                CStr(key) & _
                "}}"


            count = _
                ReplaceAllWordCI( _
                    wordDocument, _
                    token, _
                    CStr(values(key)))


            totalCount = _
                totalCount + _
                count

        End If

    Next key


    RenderWordBody = totalCount

End Function


Private Function ReplaceAllWordCI( _
    ByVal wordDocument As Object, _
    ByVal token As String, _
    ByVal replacementText As String) As Long

    Dim searchRange As Object
    Dim found As Boolean

    Dim nextStart As Long
    Dim documentEnd As Long

    Dim count As Long


    Set searchRange = _
        wordDocument.Content.Duplicate


    Do

        With searchRange.Find

            .ClearFormatting

            .Text = token

            .Forward = True
            .Wrap = WD_FIND_STOP
            .Format = False

            .MatchCase = False
            .MatchWholeWord = False
            .MatchWildcards = False

        End With


        found = _
            searchRange.Find.Execute


        If Not found Then
            Exit Do
        End If


        ' -----------------------------------------------
        ' Find成功後、searchRangeはFound Rangeになる。
        ' そのRangeだけを置換する。
        ' -----------------------------------------------
        searchRange.Text = _
            replacementText

        count = count + 1


        ' -----------------------------------------------
        ' 置換後のRange終端から再検索
        ' -----------------------------------------------
        nextStart = _
            searchRange.End

        documentEnd = _
            wordDocument.Content.End


        If nextStart >= documentEnd Then
            Exit Do
        End If


        searchRange.SetRange _
            nextStart, _
            documentEnd

    Loop


    ReplaceAllWordCI = count

End Function


' ============================================================
' Replacement verification
'
' BodyがHTMLかWord Textかに関係なく、
' Variable Tokenが消えたかを比較する。
' ============================================================

Private Function VerifyTokenReplacement( _
    ByVal originalSubject As String, _
    ByVal originalBody As String, _
    ByVal renderedSubject As String, _
    ByVal renderedBody As String, _
    ByVal values As Object) As Boolean

    Dim key As Variant
    Dim token As String

    Dim beforeCount As Long
    Dim afterCount As Long

    Dim ok As Boolean

    ok = True


    Debug.Print
    Debug.Print "----- Replacement Evidence -----"


    For Each key In values.Keys

        token = _
            "{{" & _
            CStr(key) & _
            "}}"


        beforeCount = _
            CountOccurrencesCI( _
                originalSubject, _
                token) + _
            CountOccurrencesCI( _
                originalBody, _
                token)


        afterCount = _
            CountOccurrencesCI( _
                renderedSubject, _
                token) + _
            CountOccurrencesCI( _
                renderedBody, _
                token)


        Debug.Print _
            token & _
            " : before=" & _
            beforeCount & _
            " / after=" & _
            afterCount


        If beforeCount = 0 Then
            ok = False
        End If


        If afterCount <> 0 Then
            ok = False
        End If

    Next key


    VerifyTokenReplacement = ok

End Function


' ============================================================
' Case Insensitive string helpers
' ============================================================

Private Function ReplaceAllCI( _
    ByVal sourceText As String, _
    ByVal token As String, _
    ByVal replacementText As String) As String

    ReplaceAllCI = _
        Replace( _
            sourceText, _
            token, _
            replacementText, _
            1, _
            -1, _
            vbTextCompare)

End Function


Private Function CountOccurrencesCI( _
    ByVal sourceText As String, _
    ByVal token As String) As Long

    Dim pos As Long
    Dim startPos As Long
    Dim count As Long

    startPos = 1


    Do

        pos = _
            InStr( _
                startPos, _
                sourceText, _
                token, _
                vbTextCompare)


        If pos = 0 Then
            Exit Do
        End If


        count = count + 1

        startPos = _
            pos + Len(token)

    Loop


    CountOccurrencesCI = count

End Function


' ============================================================
' TSA-006
' Route H HTML Encode
'
' Spike用最小実装。
' 本番Implementationの確定ではない。
' ============================================================

Private Function HtmlEncodeSpike( _
    ByVal value As String) As String

    Dim result As String

    result = value


    ' & は必ず最初
    result = _
        Replace(result, "&", "&amp;")

    result = _
        Replace(result, "<", "&lt;")

    result = _
        Replace(result, ">", "&gt;")

    result = _
        Replace(result, """", "&quot;")

    result = _
        Replace(result, "'", "&#39;")


    HtmlEncodeSpike = result

End Function


Private Function VerifyHtmlEncodedValue( _
    ByVal htmlBody As String, _
    ByVal rawValue As String) As Boolean

    Dim encoded As String

    encoded = _
        HtmlEncodeSpike(rawValue)


    Debug.Print
    Debug.Print "----- HTML Encode Evidence -----"
    Debug.Print "Raw     : " & rawValue
    Debug.Print "Encoded : " & encoded


    VerifyHtmlEncodedValue = _
        (InStr( _
            1, _
            htmlBody, _
            encoded, _
            vbBinaryCompare) > 0)

End Function
